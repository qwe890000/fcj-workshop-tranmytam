Bước 1: Tạo VPC

# Tạo VPC với CIDR block 10.0.0.0/16
aws ec2 create-vpc \
    --cidr-block 10.0.0.0/16 \
    --tag-specifications 'ResourceType=vpc,Tags=[{Key=Name,Value=LunaGenZ-VPC}]'

# Enable DNS hostnames
VPC_ID=$(aws ec2 describe-vpcs \
    --filters "Name=tag:Name,Values=LunaGenZ-VPC" \
    --query "Vpcs[0].VpcId" \
    --output text)

aws ec2 modify-vpc-attribute \
    --vpc-id $VPC_ID \
    --enable-dns-hostnames "{\"Value\": true}"

Bước 2: Tạo Subnets

# Public Subnet (để chạy NAT Gateway)
aws ec2 create-subnet \
    --vpc-id $VPC_ID \
    --cidr-block 10.0.1.0/24 \
    --availability-zone us-east-1a \
    --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=LunaGenZ-PublicSubnet}]'

# Private Subnet (để chạy Lambda functions)
aws ec2 create-subnet \
    --vpc-id $VPC_ID \
    --cidr-block 10.0.2.0/24 \
    --availability-zone us-east-1a \
    --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=LunaGenZ-PrivateSubnet}]'

# Lấy Subnet IDs
PUBLIC_SUBNET=$(aws ec2 describe-subnets \
    --filters "Name=tag:Name,Values=LunaGenZ-PublicSubnet" \
    --query "Subnets[0].SubnetId" --output text)

PRIVATE_SUBNET=$(aws ec2 describe-subnets \
    --filters "Name=tag:Name,Values=LunaGenZ-PrivateSubnet" \
    --query "Subnets[0].SubnetId" --output text)

echo "Public Subnet: $PUBLIC_SUBNET"
echo "Private Subnet: $PRIVATE_SUBNET"

Bước 3: Tạo Internet Gateway

# Tạo Internet Gateway
aws ec2 create-internet-gateway \
    --tag-specifications 'ResourceType=internet-gateway,Tags=[{Key=Name,Value=LunaGenZ-IGW}]'

IGW_ID=$(aws ec2 describe-internet-gateways \
    --filters "Name=tag:Name,Values=LunaGenZ-IGW" \
    --query "InternetGateways[0].InternetGatewayId" --output text)

# Attach vào VPC
aws ec2 attach-internet-gateway \
    --internet-gateway-id $IGW_ID \
    --vpc-id $VPC_ID

echo "Internet Gateway: $IGW_ID"

Bước 4: Tạo NAT Gateway

# Tạo Elastic IP
ALLOCATION_ID=$(aws ec2 allocate-address \
    --domain vpc \
    --query "AllocationId" --output text)

# Tạo NAT Gateway trong Public Subnet
aws ec2 create-nat-gateway \
    --subnet-id $PUBLIC_SUBNET \
    --allocation-id $ALLOCATION_ID \
    --tag-specifications 'ResourceType=natgateway,Tags=[{Key=Name,Value=LunaGenZ-NATGateway}]'

NAT_GW_ID=$(aws ec2 describe-nat-gateways \
    --filter "Name=tag:Name,Values=LunaGenZ-NATGateway" \
    --query "NatGateways[0].NatGatewayId" --output text)

# Đợi NAT Gateway available
echo "Đợi NAT Gateway available..."
aws ec2 wait nat-gateway-available --nat-gateway-ids $NAT_GW_ID

echo "NAT Gateway: $NAT_GW_ID"

Bước 5: Cấu hình Route Tables

# Lấy Route Table ID của VPC
ROUTE_TABLE=$(aws ec2 describe-route-tables \
    --filters "Name=vpc-id,Values=$VPC_ID" \
    --query "RouteTables[0].RouteTableId" --output text)

# Tạo Route Table riêng cho Private Subnet
PRIVATE_ROUTE_TABLE=$(aws ec2 create-route-table \
    --vpc-id $VPC_ID \
    --query "RouteTable.RouteTableId" --output text)

aws ec2 create-route \
    --route-table-id $PRIVATE_ROUTE_TABLE \
    --destination-cidr-block 0.0.0.0/0 \
    --nat-gateway-id $NAT_GW_ID

# Associate Private Subnet với Private Route Table
aws ec2 associate-route-table \
    --subnet-id $PRIVATE_SUBNET \
    --route-table-id $PRIVATE_ROUTE_TABLE

# Tạo Route Table cho Public Subnet
PUBLIC_ROUTE_TABLE=$(aws ec2 describe-route-tables \
    --filters "Name=vpc-id,Values=$VPC_ID" \
    --query "RouteTables[0].RouteTableId" --output text)

aws ec2 create-route \
    --route-table-id $PUBLIC_ROUTE_TABLE \
    --destination-cidr-block 0.0.0.0/0 \
    --gateway-id $IGW_ID

# Enable Auto-assign public IP cho Public Subnet
aws ec2 modify-subnet-attribute \
    --subnet-id $PUBLIC_SUBNET \
    --map-public-ip-on-launch

echo "Route Tables đã cấu hình xong"

Bước 6: Tạo VPC Endpoints

# VPC Endpoint cho S3 (Gateway)
aws ec2 create-vpc-endpoint \
    --vpc-id $VPC_ID \
    --service-name com.amazonaws.us-east-1.s3 \
    --route-table-ids $PRIVATE_ROUTE_TABLE \
    --tag-specifications 'ResourceType=vpc-endpoint,Tags=[{Key=Name,Value=LunaGenZ-S3-Endpoint}]'

# VPC Endpoint cho DynamoDB (Gateway)
aws ec2 create-vpc-endpoint \
    --vpc-id $VPC_ID \
    --service-name com.amazonaws.us-east-1.dynamodb \
    --route-table-ids $PRIVATE_ROUTE_TABLE \
    --tag-specifications 'ResourceType=vpc-endpoint,Tags=[{Key=Name,Value=LunaGenZ-DynamoDB-Endpoint}]'

# VPC Endpoint cho Systems Manager (Interface)
aws ec2 create-vpc-endpoint \
    --vpc-id $VPC_ID \
    --service-name com.amazonaws.us-east-1.ssm \
    --subnet-ids $PRIVATE_SUBNET \
    --vpc-endpoint-type Interface \
    --tag-specifications 'ResourceType=vpc-endpoint,Tags=[{Key=Name,Value=LunaGenZ-SSM-Endpoint}]'

echo "VPC Endpoints đã tạo xong"

Bước 7: Tạo DynamoDB Table

# Tạo DynamoDB table để lưu session
aws dynamodb create-table \
    --table-name LunaGenZ-Sessions \
    --attribute-definitions \
        AttributeName=sessionId,AttributeType=S \
    --key-schema \
        AttributeName=sessionId,KeyType=HASH \
    --billing-mode PAY_PER_REQUEST \
    --tags Key=Name,Value=LunaGenZ-Sessions

# Enable TTL
aws dynamodb update-time-to-live \
    --table-name LunaGenZ-Sessions \
    --time-to-live-specification \
        Enabled=true,AttributeName=expiresAt

echo "DynamoDB Table đã tạo xong"

Bước 8: Cấu hình Systems Manager Parameter Store

# Tạo API Key (Secure String)
aws ssm put-parameter \
    --name /lunagenz/api-key \
    --value "lunagenz-secret-key-2024" \
    --type SecureString \
    --key-id alias/aws/ssm \
    --tags Key=Name,Value=LunaGenZ-APIKey

# Tạo các parameters khác
aws ssm put-parameter \
    --name /lunagenz/dynamodb-table \
    --value "LunaGenZ-Sessions" \
    --type String

aws ssm put-parameter \
    --name /lunagenz/s3-bucket-reports \
    --value "lunagenz-reports" \
    --type String

aws ssm put-parameter \
    --name /lunagenz/external-api-url \
    --value "https://numerology-api.com" \
    --type String

echo "Parameter Store đã cấu hình xong"

Bước 9: Tạo S3 Buckets

# Tạo bucket cho static website
aws s3 mb s3://lunagenz-web --region us-east-1

# Tạo bucket cho reports
aws s3 mb s3://lunagenz-reports --region us-east-1

# Cấu hình bucket cho static website hosting
aws s3 website s3://lunagenz-web \
    --index-document index.html \
    --error-document error.html

# Cấu hình bucket policy cho public read (web bucket)
cat > web-bucket-policy.json << 'EOF'
{
    "Version": "2012-10-17",
    "Statement": [
        {
            "Sid": "PublicReadGetObject",
            "Effect": "Allow",
            "Principal": "*",
            "Action": "s3:GetObject",
            "Resource": "arn:aws:s3:::lunagenz-web/*"
        }
    ]
}
EOF

aws s3api put-bucket-policy \
    --bucket lunagenz-web \
    --policy file://web-bucket-policy.json

# Cấu hình CORS cho web bucket
aws s3api put-bucket-cors \
    --bucket lunagenz-web \
    --cors-configuration '{
        "CORSRules": [
            {
                "AllowedHeaders": ["*"],
                "AllowedMethods": ["GET"],
                "AllowedOrigins": ["*"],
                "MaxAgeSeconds": 3000
            }
        ]
    }'

echo "S3 Buckets đã tạo và cấu hình xong"

Bước 10: Tạo IAM Roles

Tạo Execution Role cho Lambda

# Tạo Role
aws iam create-role \
    --role-name LunaGenZ-LambdaExecutionRole \
    --assume-role-policy-document '{
        "Version": "2012-10-17",
        "Statement": [
            {
                "Effect": "Allow",
                "Principal": {"Service": "lambda.amazonaws.com"},
                "Action": "sts:AssumeRole"
            }
        ]
    }'

# Đợi role được tạo
sleep 10

# Attach policies
aws iam attach-role-policy \
    --role-name LunaGenZ-LambdaExecutionRole \
    --policy-arn arn:aws:iam::aws:policy/service-role/AWSLambdaBasicExecutionRole

aws iam attach-role-policy \
    --role-name LunaGenZ-LambdaExecutionRole \
    --policy-arn arn:aws:iam::aws:policy/AmazonDynamoDBFullAccess

aws iam attach-role-policy \
    --role-name LunaGenZ-LambdaExecutionRole \
    --policy-arn arn:aws:iam::aws:policy/AmazonS3FullAccess

aws iam attach-role-policy \
    --role-name LunaGenZ-LambdaExecutionRole \
    --policy-arn arn:aws:iam::aws:policy/AmazonSSMReadOnlyAccess

echo "IAM Execution Role đã tạo xong"

Bước 11: Thiết lập Security Group cho Lambda

# Tạo Security Group cho Lambda functions
aws ec2 create-security-group \
    --group-name LunaGenZ-Lambda-SG \
    --description "Security group for LunaGenZ Lambda functions" \
    --vpc-id $VPC_ID

LAMBDA_SG=$(aws ec2 describe-security-groups \
    --filters "Name=group-name,Values=LunaGenZ-Lambda-SG" \
    --query "SecurityGroups[0].GroupId" --output text)

# Cho phép outbound traffic
aws ec2 authorize-security-group-egress \
    --group-id $LAMBDA_SG \
    --ip-permissions IpProtocol=all,IpRanges='[{CidrIp=0.0.0.0/0}]'

echo "Security Group: $LAMBDA_SG"
