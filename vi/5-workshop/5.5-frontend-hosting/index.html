js/config.js

// Cấu hình API - THAY ĐỔI THEO API CỦA BẠN
const CONFIG = {
    API_BASE_URL: 'https://YOUR_API_ID.execute-api.us-east-1.amazonaws.com/prod',
    
    // Các endpoints
    ENDPOINTS: {
        CALCULATE: '/calculate',
        GENERATE_REPORT: '/generate-report'
    },
    
    // Cấu hình khác
    TIMEOUT: 30000, // 30 seconds
    PDF_EXPIRY_HOURS: 1
};

js/app.js

// LunaGenZ Frontend Application

class LunaGenZApp {
    constructor() {
        this.form = document.getElementById('numerologyForm');
        this.resultDiv = document.getElementById('result');
        this.errorDiv = document.getElementById('error');
        this.loadingDiv = document.getElementById('loading');
        this.submitBtn = document.getElementById('submitBtn');
        
        this.currentData = null;
        
        this.init();
    }

    init() {
        // Bind event listeners
        this.form.addEventListener('submit', (e) => this.handleSubmit(e));
        document.getElementById('downloadPdfBtn').addEventListener('click', () => this.downloadPDF());
    }

    async handleSubmit(e) {
        e.preventDefault();

        const fullName = document.getElementById('fullName').value.trim();
        const birthDate = document.getElementById('birthDate').value;

        if (!fullName || !birthDate) {
            this.showError('Vui lòng điền đầy đủ thông tin');
            return;
        }

        await this.calculateNumerology(fullName, birthDate);
    }

    async calculateNumerology(fullName, birthDate) {
        this.setLoading(true);
        this.hideError();
        this.hideResult();

        try {
            const response = await fetch(CONFIG.API_BASE_URL + CONFIG.ENDPOINTS.CALCULATE, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    fullName,
                    birthDate
                })
            });

            const data = await response.json();

            if (response.ok && data.sessionId) {
                this.currentData = data;
                this.displayResult(data);
            } else {
                this.showError(data.error || 'Đã xảy ra lỗi khi tính toán');
            }
        } catch (error) {
            console.error('Error:', error);
            this.showError('Không thể kết nối đến server. Vui lòng thử lại sau.');
        } finally {
            this.setLoading(false);
        }
    }

    displayResult(data) {
        // Hiển thị thông tin người dùng
        document.getElementById('resultName').textContent = data.fullName;
        document.getElementById('resultDate').textContent = data.birthDate;

        // Hiển thị các con số
        document.getElementById('lifePathNumber').textContent = data.numerology.lifePath.number;
        document.getElementById('lifePathDesc').textContent = data.numerology.lifePath.description;

        document.getElementById('expressionNumber').textContent = data.numerology.expression.number;
        document.getElementById('expressionDesc').textContent = data.numerology.expression.description;

        document.getElementById('soulUrgeNumber').textContent = data.numerology.soulUrge.number;
        document.getElementById('soulUrgeDesc').textContent = data.numerology.soulUrge.description;

        this.showResult();
    }

    async downloadPDF() {
        if (!this.currentData) {
            this.showError('Vui lòng tính toán trước khi tải PDF');
            return;
        }

        const btn = document.getElementById('downloadPdfBtn');
        const originalText = btn.innerHTML;
        btn.innerHTML = '⏳ Đang tạo PDF...';
        btn.disabled = true;

        try {
            const response = await fetch(CONFIG.API_BASE_URL + CONFIG.ENDPOINTS.GENERATE_REPORT, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(this.currentData)
            });

            const data = await response.json();

            if (data.success && data.downloadUrl) {
                // Mở URL để tải PDF
                window.open(data.downloadUrl, '_blank');
            } else {
                this.showError('Không thể tạo báo cáo PDF');
            }
        } catch (error) {
            console.error('PDF Error:', error);
            this.showError('Lỗi khi tải báo cáo: ' + error.message);
        } finally {
            btn.innerHTML = originalText;
            btn.disabled = false;
        }
    }

    showResult() {
        this.resultDiv.style.display = 'block';
        this.resultDiv.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

    hideResult() {
        this.resultDiv.style.display = 'none';
    }

    showError(message) {
        document.getElementById('errorMessage').textContent = message;
        this.errorDiv.style.display = 'block';
    }

    hideError() {
        this.errorDiv.style.display = 'none';
    }

    setLoading(loading) {
        this.submitBtn.disabled = loading;
        
        if (loading) {
            this.form.querySelector('.btn-text').style.display = 'none';
            this.form.querySelector('.btn-loading').style.display = 'inline';
        } else {
            this.form.querySelector('.btn-text').style.display = 'inline';
            this.form.querySelector('.btn-loading').style.display = 'none';
        }
    }
}

// Initialize app when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
    window.app = new LunaGenZApp();
});

Bước 6: Tạo CloudFront Distribution

# Tạo CloudFront Distribution
DISTRIBUTION_ID=$(aws cloudfront create-distribution \
    --origin-domain-name lunagenz-web.s3.amazonaws.com \
    --default-root-object index.html \
    --query 'Distribution.Id' \
    --output text)

# Lấy domain name
DOMAIN_NAME=$(aws cloudfront get-distribution \
    --id $DISTRIBUTION_ID \
    --query 'Distribution.DomainName' \
    --output text)

echo "CloudFront Domain: $DOMAIN_NAME"
echo "Distribution ID: $DISTRIBUTION_ID"

Thiết lập Custom Domain (Tùy chọn)

Chuẩn bị Domain

  1. Mua domain từ Route 53 hoặc registrar khác
  2. Tạo hosted zone trong Route 53
  3. Thêm CNAME record trỏ đến CloudFront
# Tạo hosted zone
aws route53 create-hosted-zone \
    --name lunagenz.app \
    --caller-reference $(date +%s)

# Tạo CNAME record
aws route53 change-resource-record-sets \
    --hosted-zone-id ZONE_ID \
    --change-batch '{
        "Changes": [{
            "Action": "CREATE",
            "ResourceRecordSet": {
                "Name": "lunagenz.app",
                "Type": "CNAME",
                "TTL": 300,
                "ResourceRecords": [{ "Value": "'$DOMAIN_NAME'" }]
            }
        }]
    }'
